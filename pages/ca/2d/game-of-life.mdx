import PostLayout from '@components/PostLayout'
import CA2D from '@components/CA2D'
import { any, eq, sum } from '@lancejpollard/ca'

export const meta = {
  url: '/ca/2d/game-of-life',
  tab: `Conway's Game of Life`,
  note: `This is Conway's Game of Life, a 2D cellular automaton, implemented in TypeScript.`,
  tags: ['cellular automata'],
}

export const caProps = {
  h: 60,
  max: 1,
  move: [
    n => {
      const t = sum(n)
      const isLive = n[4] === 1
      if (isLive && t >= 3 && t <= 4) {
        return 1
      } else {
        return -1
      }
    },
    n => {
      const t = sum(n)
      const isDead = n[4] === 0
      if (isDead && t === 3) {
        return 1
      } else {
        return -1
      }
    },
    n => {
      const isLive = n[4] === 1
      return isLive ? 0 : -1
    },
  ],
  w: 60,
}

## Automaton

<CA2D
  {...caProps}
  steps={10000}
/>

## Notes

This is Conway's Game of Life, a 2D cellular automaton, implemented in
TypeScript.

It is interesting because it shows that a 2D CA can have interesting,
life-like patterns given only 3 rules:

1. Any live cell with two or three live neighbours survives.
1. Any dead cell with three live neighbours becomes a live cell.
1. All other live cells die in the next generation. Similarly, all other
   dead cells stay dead.

export default ({ children }) => (
  <PostLayout {...meta}>{children}</PostLayout>
)
